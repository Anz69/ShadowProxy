// Получение реферального баланса
// GET /user-api/balance
// Примечание: реферальный баланс возвращается вместе с основным балансом в поле referralBalance

const apiKey = 'YOUR_API_KEY';
const url = 'https://shadowproxy.io/api/user-api/balance';

fetch(url, {
  method: "GET",
  headers: {
    "api-key": apiKey,
    "Content-Type": "application/json"
  }
})
.then(response => response.json())
.then(data => {
  console.log('Balance:', data.message.balance);
  console.log('Referral Balance:', data.message.referralBalance);
})
.catch(error => console.error('Request failed:', error));
