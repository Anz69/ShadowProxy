// Получение списка тарифов с ценами
// GET /user-api/tariffs

const apiKey = 'YOUR_API_KEY';
const url = 'https://shadowproxy.io/api/user-api/tariffs';

fetch(url, {
  method: "GET",
  headers: {
    "api-key": apiKey,
    "Content-Type": "application/json"
  }
})
.then(response => response.json())
.then(data => {
  console.log('Tariffs:', data.message);
  // Массив тарифов с ценами за ГБ и доступными пакетами
})
.catch(error => console.error('Request failed:', error));
