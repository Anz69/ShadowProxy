# Получение списка тарифов с ценами
# GET /user-api/tariffs

import requests

api_key = 'YOUR_API_KEY'
url = 'https://shadowproxy.io/api/user-api/tariffs'

headers = {
    "api-key": api_key,
    "Content-Type": "application/json"
}

response = requests.get(url, headers=headers)
data = response.json()
print('Tariffs:', data['message'])
