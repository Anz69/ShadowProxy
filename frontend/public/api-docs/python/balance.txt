# Получение баланса и реферального баланса
# GET /user-api/balance

import requests

api_key = 'YOUR_API_KEY'
url = 'https://shadowproxy.io/api/user-api/balance'

headers = {
    "api-key": api_key,
    "Content-Type": "application/json"
}

response = requests.get(url, headers=headers)
data = response.json()
print('Response:', data)
