# Получение списка доступных стран для пакета
# GET /user-api/countries?packageType=RESIDENT
# Для MOBILE типа возвращает также список операторов по странам

import requests

api_key = 'YOUR_API_KEY'
package_type = 'RESIDENT'
url = f'https://shadowproxy.io/api/user-api/countries?packageType={package_type}'

headers = {
    "api-key": api_key,
    "Content-Type": "application/json"
}

response = requests.get(url, headers=headers)
data = response.json()
print('Response:', data)
