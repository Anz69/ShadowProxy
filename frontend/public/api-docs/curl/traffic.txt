# Получение истории трафика
# GET /user-api/traffic?packageId=package-uuid (по пакету за неделю)
# GET /user-api/traffic (по всему аккаунту)

# По пакету за неделю:
curl -X GET "https://shadowproxy.io/api/user-api/traffic?packageId=package-uuid&week=1" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json"

# По всему аккаунту:
curl -X GET "https://shadowproxy.io/api/user-api/traffic" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json"
