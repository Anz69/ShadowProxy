# Создание прокси на пакет
# POST /user-api/packages/:packageId/proxies
# Максимум 1000 активных прокси на аккаунт

curl -X POST "https://shadowproxy.io/api/user-api/packages/package-uuid-here/proxies" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "location": "US",
    "quantity": 1,
    "rotation": "BY_REQUEST"
  }'

# Для ротации по времени:
curl -X POST "https://shadowproxy.io/api/user-api/packages/package-uuid-here/proxies" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "location": "US",
    "quantity": 1,
    "rotation": "BY_TIME",
    "rotationInterval": 30
  }'
