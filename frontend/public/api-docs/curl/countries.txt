# Получение списка доступных стран для пакета
# GET /user-api/countries?packageType=RESIDENT
# Для MOBILE типа возвращает также список операторов по странам

curl -X GET "https://shadowproxy.io/api/user-api/countries?packageType=RESIDENT" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json"

# Успешный ответ (для типов кроме MOBILE):
# {
#   "status": "ok",
#   "code": "200",
#   "message": ["us", "de", "nl", "gb", "fr", ...]
# }
