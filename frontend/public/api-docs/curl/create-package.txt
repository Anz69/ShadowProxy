# Создание пакета по типу
# POST /user-api/packages
# Пакеты по ГБ создаются бесплатно, мобильные по времени требуют оплаты

curl -X POST "https://shadowproxy.io/api/user-api/packages" \
  -H "api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "type": "RESIDENT"
  }'

# Успешный ответ:
# {
#   "status": "ok",
#   "code": "200",
#   "message": {
#     "id": "uuid-here",
#     "userId": "user-uuid",
#     "type": "RESIDENT",
#     "createdAt": "2026-02-05T12:00:00.000Z",
#     "updatedAt": "2026-02-05T12:00:00.000Z"
#   }
# }

# Ошибка (пакет уже существует):
# {
#   "status": "error",
#   "code": "PACKAGE_ALREADY_EXISTS",
#   "message": "Пакет этого типа уже активирован"
# }

# Ошибка (неверный тип пакета):
# {
#   "status": "error",
#   "code": "INVALID_PACKAGE_TYPE",
#   "message": "Неизвестный тип пакета"
# }
